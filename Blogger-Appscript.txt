      xmlns:expr='http://www.google.com/2005/gml/expr'>

<head>

  <meta content='width=device-width, initial-scale=1' name='viewport'/>

  <title><data:blog.pageTitle/></title>

  <!-- ========================================= -->
  <!-- REDIRECT HP KE MODE DESKTOP BLOGGER      -->
  <!-- ========================================= -->

  <script type='text/javascript'>
  //<![CDATA[

  (function () {

    var isMobile =
      /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i
      .test(navigator.userAgent);

    var url = new URL(window.location.href);

    /*
     * Jika dibuka dari HP dan Blogger memberikan ?m=1,
     * hapus parameter tersebut agar menggunakan mode desktop.
     */
    if (isMobile && url.searchParams.get('m') === '1') {

      url.searchParams.delete('m');

      window.location.replace(url.toString());

    }

  })();

  //]]>
  </script>


  <!-- ========================================= -->
  <!-- MEMBERSIHKAN PARAMETER m=1 / m=0         -->
  <!-- ========================================= -->

  <script type='text/javascript'>
  //<![CDATA[

  (function () {

    var url = new URL(window.location.href);

    if (
      url.searchParams.has('m')
    ) {

      url.searchParams.delete('m');

      window.history.replaceState(
        null,
        '',
        url.pathname +
        (url.searchParams.toString()
          ? '?' + url.searchParams.toString()
          : '') +
        url.hash
      );

    }

  })();

  //]]>
  </script>

    body {

      height: 100%;

  ]]></b:skin>
